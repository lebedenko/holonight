cmake_minimum_required(VERSION 3.25)
project(preview_benchmark LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(HolonightFileBrowser REQUIRED CONFIG COMPONENTS Preview)
add_executable(preview-benchmark main.cpp)
target_link_libraries(preview-benchmark PRIVATE HolonightFileBrowser::Preview)
target_compile_features(preview-benchmark PRIVATE cxx_std_23)
option(ISOLATED "Use the explicit runtime constructor" ON)
if(ISOLATED)
  target_compile_definitions(preview-benchmark PRIVATE ISOLATED)
endif()

option(BUILD_GRID_CONTROL "Build the real installed ThumbnailProvider control" OFF)
if(BUILD_GRID_CONTROL)
  find_package(HolonightFileBrowser REQUIRED CONFIG COMPONENTS PreviewQuick)
  add_executable(preview-grid-control grid.cpp)
  target_link_libraries(preview-grid-control PRIVATE HolonightFileBrowser::PreviewQuick)
  target_compile_features(preview-grid-control PRIVATE cxx_std_23)
  if(ISOLATED)
    target_compile_definitions(preview-grid-control PRIVATE ISOLATED)
  endif()
endif()
if(BUILD_GRID_CONTROL)
  enable_testing()
  add_executable(preview-grid-test grid_test.cpp)
  target_link_libraries(preview-grid-test PRIVATE HolonightFileBrowser::PreviewQuick)
  target_compile_features(preview-grid-test PRIVATE cxx_std_23)
  add_test(NAME scheduler-barriers COMMAND preview-grid-test)
endif()

set(PREVIEW_DIAGNOSTIC_HEADER "" CACHE FILEPATH "Private overlay header, diagnostic build only")
if(PREVIEW_DIAGNOSTIC_HEADER)
  target_compile_definitions(preview-benchmark PRIVATE PREVIEW_DIAGNOSTIC_HEADER="${PREVIEW_DIAGNOSTIC_HEADER}")
endif()
