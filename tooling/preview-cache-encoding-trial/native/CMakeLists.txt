cmake_minimum_required(VERSION 3.25)
project(cache_encoding_investigation LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Gui)
find_package(HolonightImages CONFIG REQUIRED)
foreach(mode IN ITEMS default 11 0)
  add_library(store_${mode} STATIC ${CMAKE_CURRENT_SOURCE_DIR}/../sources/${mode}/src/cache.cpp)
  target_include_directories(store_${mode} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../sources/${mode}/include)
  target_compile_features(store_${mode} PRIVATE cxx_std_23)
  target_compile_definitions(store_${mode} PRIVATE HolonightThumbnails=Store_${mode})
  target_link_libraries(store_${mode} PRIVATE Qt6::Gui)
endforeach()
add_executable(investigate main.cpp)
target_compile_features(investigate PRIVATE cxx_std_23)
target_include_directories(investigate PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../sources/default/include)
target_link_libraries(investigate PRIVATE store_default store_11 store_0 HolonightImages::Images Qt6::Gui)
add_library(mapping SHARED mapping.cpp)
target_link_libraries(mapping PRIVATE dl)
