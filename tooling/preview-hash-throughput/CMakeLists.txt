cmake_minimum_required(VERSION 3.24)
project(preview_hash_throughput LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(preview-hash-throughput main.cpp)
target_link_libraries(preview-hash-throughput PRIVATE Qt6::Core OpenSSL::Crypto)
target_compile_options(preview-hash-throughput PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME correctness COMMAND preview-hash-throughput --self-test)
