cmake_minimum_required(VERSION 3.24)
project(preview_hash_device_mechanism LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(preview-hash-device-mechanism main.cpp)
target_link_libraries(preview-hash-device-mechanism PRIVATE Qt6::Core)
target_compile_options(preview-hash-device-mechanism PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME correctness COMMAND preview-hash-device-mechanism --self-test)
